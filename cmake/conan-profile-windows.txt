# Trussline Conan host profile (Windows / MSVC).
#
# Matches the compiler that ACTUALLY builds this project: cl 19.44 from
# Visual Studio Build Tools 2022 (_MSC_VER 1944 -> Conan compiler.version=194).
#
# `conan profile detect` gets this WRONG on this machine. It reports 195,
# because vswhere finds "Visual Studio Community 2026" (v18.7) - an install
# that carries NO C++ toolset at all (confirmed: vswhere -requires
# VC.Tools.x86.x64 lists only the 2022 and 2019 Build Tools). Conan then tries
# to drive a toolchain that cannot compile anything:
#
#   compiler.version=195  ->  CMake Error: Could not create named generator
#                             "Visual Studio 18 2026"
#
# Two other values were tried and are worse - recorded so they are not retried:
#
#   compiler.version=193  ->  Conan requests vcvars_ver=14.3, but this machine
#                             has a 14.4x toolset:
#                             "Toolset directory for version '14.3' was not found"
#                             Every from-source dependency build dies.
#
# compiler.version is not a cosmetic package-ID label: Conan derives both the
# vcvars toolset request and the CMake generator from it. It must describe the
# compiler that is really present.

[settings]
os=Windows
arch=x86_64
build_type=Release
compiler=msvc
compiler.version=194
# cppstd=17, not Conan's default 14. The RoR remote publishes its prebuilt
# binaries at cppstd=17/20/23; at 14 almost nothing matches and Conan falls
# back to building from source. That matters beyond speed: MyGUI 3.4.1/3.4.3
# CANNOT be built from source (their recipe references a
# CMake/Utils/PrecompiledHeader.cmake absent from the tarball), and MyGUI 3.4.0
# - the one that does build - is incompatible with OGRE 14:
#
#   MyGUI_OgreRenderManager.cpp(371): error C2039:
#     '_setTextureAddressingMode': is not a member of 'Ogre::RenderSystem'
#
# So reaching the published 3.4.x binaries is the only route to a MyGUI that
# works with OGRE 14. Unlike the compiler.version pin tried earlier, this is a
# genuine build choice rather than a misdescription of the toolchain: RoR's
# MSVC path sets no language standard, and C++17 is a superset of what it uses.
compiler.cppstd=17
compiler.runtime=dynamic
compiler.runtime_type=Release
