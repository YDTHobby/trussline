# Trussline Conan host profile (Windows / MSVC).
#
# Why this exists rather than using Conan's auto-detected default:
#
# The RoR dependency remote publishes prebuilt binaries for MSVC
# compiler.version=193. Auto-detection on a current VS 2022 install reports
# 195, so Conan finds no matching binary and falls back to building from
# source - which fails, because the MyGUI recipe's build() references
# CMake/Utils/PrecompiledHeader.cmake, a file its source tarball does not
# contain (FileNotFoundError, reproduced on both 3.4.1 and 3.4.3).
#
# 193 and 195 are both VS 2022 on the SAME v143 toolset and are ABI-compatible,
# so this only steers Conan's package-ID resolution toward the published
# binaries. The actual compiler used is still the real cl.exe from vcvars -
# nothing is misbuilt, and the binaries we pull were produced by a compatible
# toolset.
#
# Revisit when: the remote publishes 195 binaries, or the MyGUI recipe is
# fixed upstream, or MyGUI is removed entirely (Phase 5 replaces it for touch).

[settings]
os=Windows
arch=x86_64
build_type=Release
compiler=msvc
compiler.version=193
compiler.cppstd=14
compiler.runtime=dynamic
compiler.runtime_type=Release
