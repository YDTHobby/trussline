# Trussline Conan host profile (Windows / MSVC).
#
# Matches the ACTUAL installed toolchain. Do not pin compiler.version to reach
# older prebuilt binaries - that was tried and it fails:
#
#   compiler.version=193  ->  Conan passes vcvars_ver=14.3 to conanvcvars.bat
#                         ->  "Toolset directory for version '14.3' was not found"
#
# because this machine has MSVC 14.44 (a 14.4x toolset). The version setting is
# not merely a package-ID label; Conan derives the vcvars toolset request from
# it, so lying about it breaks every from-source dependency build.
#
# The consequence is that dependencies without a matching prebuilt binary get
# built from source. That is slower but correct, and it is the reason the MyGUI
# recipe bug matters (see conanfile.py).

[settings]
os=Windows
arch=x86_64
build_type=Release
compiler=msvc
compiler.version=195
compiler.cppstd=14
compiler.runtime=dynamic
compiler.runtime_type=Release
